Blocos de código

```C
{
//tudo oque estiver aqui estara em um bloco de código
}
```

eles podem ser utilizados no ```if(){}else{} ``` e ```while(){} for(){}```


verificar positivo ou negativo:

```C
#include <stdio.h>

int main(void){

    unsigned char a = -4;

    unsigned char positivo = (a >> 7);

    //   10000000
    // & 10000100
    //   10000101
    if(positivo)
        printf("N");
    return 0;
}

```


Utilizando o switch:

```C
#include <stdio.h>

int main(void){

    unsigned int a = 1;

    scanf("%d", &a);

    switch(a){
        case 1: 
            printf("Mes %u", a);
            break;
        case 2: 
            printf("Mes %u", a);
            break;
        case 3: 
            printf("Mes %u", a);
            break;
        case 4: 
            printf("Mes %u", a);
            break;
        case 5: 
            printf("Mes %u", a);
            break;

        default: 
            printf("Mes invalido");
            break; //não precisa do break; no default
    }

    return 0;
}

```
